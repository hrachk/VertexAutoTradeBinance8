 Автоматическая торговая стратегия возврата к среднему, основанная на количественном анализе. Покупает на сильных просадках и продаёт на пиках, используя отклонения от нормы и подтверждение объёмами.
Алгоритм использует набор статистических индикаторов для выявления локальных экстремумов — максимумов и минимумов колебаний цены. Основная логика стратегии основана на анализе отклонений: такие индикаторы, как Z-оценка (Z-Score), полосы стандартного отклонения и модифицированные конверты Боллинджера (Modified Bollinger Enevelopes), рассчитывают, насколько текущая цена отклоняется от своего статистического среднего на основе недавних данных.

Пример:
Если цена монеты опускается более чем на 2.0 стандартных отклонения (Z-Score) ниже своего скользящего среднего, алгоритм помечает это как статистически значимую аномалию.

Далее стратегия ищет подтверждение в торговом объёме — например, всплеск объёма выше среднего за 50 периодов — чтобы убедиться, что движение не связано с рыночным шумом или низкой ликвидностью.

Если сигнал подтверждается, система открывает позицию на возврат к среднему, рассчитывая, что цена вернётся к своему усреднённому значению (например, от 2σ к 0σ).

В некоторых случаях, при наличии сильного импульса, алгоритм может удерживать позицию дольше — до момента, когда цена отклонится в обратную сторону, например до +0.5σ или +1σ выше среднего, прежде чем выйти из сделки.

По сути, бот стремится покупать на фоне паники и продавать при эйфории, опираясь не на субъективные графические паттерны, а на количественно измеримые отклонения от ожидаемого статистического поведения.

 private record UniverseSnapshot(
        DateTime UtcTime,
        IReadOnlyList<string> All,
        IReadOnlyList<string> Long,
        IReadOnlyList<string> Short,
        IReadOnlyList<string> Pinned,
        IReadOnlyList<string> PinnedByPositions,
        int DynamicCap,
        int AiCapLong,
        int AiCapShort,
        decimal BtcVol,
        string BtcVolBucket)
    {
        public static UniverseSnapshot Empty() => new(
            DateTime.MinValue,
            Array.Empty<string>(),
            Array.Empty<string>(),
            Array.Empty<string>(),
            Array.Empty<string>(),
            Array.Empty<string>(),
            0, 0, 0,
            0m, "NA");
    }

    private static decimal Normalize(decimal value, decimal min, decimal max)
    {
        if (max <= min) return 0.5m;
        return Math.Clamp((value - min) / (max - min), 0m, 1m);
    }

    private static IReadOnlyList<string> BlacklistFilter(IEnumerable<string> symbols)
        => symbols.Where(s => !string.Equals(s, "AIAUSDT", StringComparison.OrdinalIgnoreCase)).ToList();

    private static IReadOnlyList<string> NormalizeSymbols(IEnumerable<string> symbols)
        => symbols
            .Where(s => !string.IsNullOrWhiteSpace(s))
            .Select(s => s.Trim().ToUpperInvariant())
            .Distinct(StringComparer.OrdinalIgnoreCase)
            .ToList();

    private static IReadOnlyList<string> LimitUnion(
       IEnumerable<string> pinnedCfg,
       IEnumerable<string> pinnedByPos,
       IEnumerable<string> longs,
       IEnumerable<string> shorts,
       int totalCap)
    {
        var pinCfg = NormalizeSymbols(pinnedCfg);
        var pinPos = NormalizeSymbols(pinnedByPos); // Открытые позиции

        // 1. Позиции — абсолютный приоритет, не учитываются в лимите totalCap
        var result = new List<string>(pinPos);

        // 2. Добавляем Pinned из конфига, если есть место до totalCap
        var otherCandidates = pinCfg
            .Concat(NormalizeSymbols(longs))
            .Concat(NormalizeSymbols(shorts))
            .Distinct(StringComparer.OrdinalIgnoreCase)
            .Where(s => !result.Contains(s, StringComparer.OrdinalIgnoreCase));

        // Заполняем до totalCap, но pinPos уже там (даже если их > totalCap)
        int slotsForOthers = Math.Max(0, totalCap - result.Count);
        result.AddRange(otherCandidates.Take(slotsForOthers));

        return result.Distinct(StringComparer.OrdinalIgnoreCase).ToList();
    }
